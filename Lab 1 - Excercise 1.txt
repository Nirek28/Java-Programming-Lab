import java.util.Scanner;

public class Main {
    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        int id = sc.nextInt();
        String name = sc.next();

        int present = 0;
        int absent = 0;

        for (int i = 0; i < 7; i++) {

            int x = sc.nextInt();

            if (x == 1) {
                present++;
            }
            else if (x == 0) {
                absent++;
            }
            else {
                System.out.println("Invalid Attendance Input");
                return;
            }
        }

        double percentage = present * 100.0 / 7;

        if (percentage >= 90) {
            System.out.printf("Attendance = %.2f%%, Absent Days = %d, Eligible",
                    percentage, absent);
        }
        else {
            System.out.printf("Attendance = %.2f%%, Absent Days = %d, Not Eligible",
                    percentage, absent);
        }
    }
}
